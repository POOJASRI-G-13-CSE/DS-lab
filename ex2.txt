#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

struct Node *last = NULL;

// Insert element at the end
void insert(int value)
{
    struct Node *newNode;

    newNode = (struct Node *)malloc(sizeof(struct Node));

    newNode->data = value;

    if (last == NULL)
    {
        last = newNode;
        newNode->next = last;
    }
    else
    {
        newNode->next = last->next;
        last->next = newNode;
        last = newNode;
    }

    printf("Element inserted successfully.\n");
}

// Delete element
void deleteElement(int value)
{
    struct Node *temp, *prev;

    if (last == NULL)
    {
        printf("List is empty!\n");
        return;
    }

    temp = last->next;
    prev = last;

    do
    {
        if (temp->data == value)
        {
            if (temp == last && temp->next == last)
            {
                last = NULL;
            }
            else
            {
                prev->next = temp->next;

                if (temp == last)
                    last = prev;
            }

            free(temp);
            printf("Element deleted successfully.\n");
            return;
        }

        prev = temp;
        temp = temp->next;

    } while (temp != last->next);

    printf("Element not found!\n");
}

// Search element
void search(int value)
{
    struct Node *temp;
    int position = 0;

    if (last == NULL)
    {
        printf("List is empty!\n");
        return;
    }

    temp = last->next;

    do
    {
        if (temp->data == value)
        {
            printf("Element found at position %d.\n", position);
            return;
        }

        temp = temp->next;
        position++;

    } while (temp != last->next);

    printf("Element not found!\n");
}

// Display circular list
void display()
{
    struct Node *temp;

    if (last == NULL)
    {
        printf("List is empty!\n");
        return;
    }

    temp = last->next;

    printf("Circular List: ");

    do
    {
        printf("%d -> ", temp->data);
        temp = temp->next;

    } while (temp != last->next);

    printf("(back to first element)\n");
}

int main()
{
    int choice, value;

    while (1)
    {
        printf("\n--- CIRCULAR LIST ADT ---\n");
        printf("1. Insert\n");
        printf("2. Delete\n");
        printf("3. Search\n");
        printf("4. Display\n");
        printf("5. Exit\n");

        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                printf("Enter value: ");
                scanf("%d", &value);
                insert(value);
                break;

            case 2:
                printf("Enter value to delete: ");
                scanf("%d", &value);
                deleteElement(value);
                break;

            case 3:
                printf("Enter value to search: ");
                scanf("%d", &value);
                search(value);
                break;

            case 4:
                display();
                break;

            case 5:
                printf("Program terminated.\n");
                exit(0);

            default:
                printf("Invalid choice!\n");
        }
    }

    return 0;
}
