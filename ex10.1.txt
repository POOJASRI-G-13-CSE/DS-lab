#include <stdio.h>

int main()
{
    int arr[100], n, i, key;
    int found = 0;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter %d elements:\n", n);

    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
    }

    printf("Enter element to search: ");
    scanf("%d", &key);

    // Linear Search
    for (i = 0; i < n; i++)
    {
        if (arr[i] == key)
        {
            printf("Element %d found at position %d.\n", key, i + 1);
            found = 1;
            break;
        }
    }

    if (found == 0)
    {
        printf("Element %d not found in the array.\n", key);
    }

    return 0;
}
