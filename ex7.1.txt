#include <stdio.h>

#define MAX 20

int queue[MAX];
int front = -1;
int rear = -1;
int visited[MAX];

void BFS(int graph[MAX][MAX], int n, int start)
{
    int i, current;

    // Insert starting vertex into queue
    queue[++rear] = start;
    visited[start] = 1;

    while (front != rear)
    {
        current = queue[++front];

        printf("%d ", current);

        // Check all adjacent vertices
        for (i = 0; i < n; i++)
        {
            if (graph[current][i] == 1 && visited[i] == 0)
            {
                queue[++rear] = i;
                visited[i] = 1;
            }
        }
    }
}

int main()
{
    int graph[MAX][MAX];
    int n, start;
    int i, j;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter adjacency matrix:\n");

    for (i = 0; i < n; i++)
    {
        for (j = 0; j < n; j++)
        {
            scanf("%d", &graph[i][j]);
        }
    }

    printf("Enter starting vertex (0 to %d): ", n - 1);
    scanf("%d", &start);

    for (i = 0; i < n; i++)
        visited[i] = 0;

    printf("\nBFS Traversal: ");

    BFS(graph, n, start);

    printf("\n");

    return 0;
}
